Option Explicit

' Paste into the Bridge worksheet code module.
' Each BridgeQueue row has a Trigger formula =[@CommandID]; calculation Automatic.
' BridgeQueue is an Excel Table on Bridge. Each row is a durable command
' only after the workbook is saved. This prototype writes to Data in this workbook.
Private Sub Worksheet_Calculate()
    Static busy As Boolean
    If busy Then Exit Sub
    busy = True
    On Error GoTo SafeExit
    ProcessQueue
SafeExit:
    busy = False
End Sub

Private Sub ProcessQueue()
    Dim queue As ListObject, data As ListObject, job As ListRow
    Set queue = Me.ListObjects("BridgeQueue")
    Set data = ThisWorkbook.Worksheets("Data").ListObjects("DischargeRecords")
    For Each job In queue.ListRows
        If CStr(GetField(queue, job, "Status")) = "PENDING" Then
            On Error GoTo JobFailed
            If CStr(GetField(queue, job, "Action")) <> "ADD" Then _
                Err.Raise vbObjectError + 210, , "Unsupported action"
            AddRecord queue, job, data
            SetField queue, job, "Result", "Saved to DischargeRecords"
            SetField queue, job, "Status", "DONE"
            On Error GoTo 0
        End If
ContinueJobs:
    Next job
    Exit Sub
JobFailed:
    Dim errorMessage As String
    errorMessage = Err.Description
    Err.Clear
    On Error Resume Next
    SetField queue, job, "Result", errorMessage
    SetField queue, job, "Status", "ERROR"
    On Error GoTo 0
    Resume ContinueJobs
End Sub

Private Sub AddRecord(ByVal q As ListObject, ByVal job As ListRow, ByVal t As ListObject)
    Dim patientId As String, admissionDate As String, recordId As String
    patientId = CStr(GetField(q, job, "PatientID"))
    admissionDate = CStr(GetField(q, job, "AdmissionDate"))
    recordId = CStr(GetField(q, job, "RecordID"))
    If Len(patientId) = 0 Or Len(admissionDate) = 0 Or Len(recordId) = 0 Then _
        Err.Raise vbObjectError + 211, , "Missing patient, admission, or record ID"

    Dim r As ListRow, source As ListRow
    For Each r In t.ListRows
        If CStr(GetField(t, r, "RecordID")) = recordId Then Exit Sub
        If CStr(GetField(t, r, "PatientID")) = patientId And _
           CStr(GetField(t, r, "AdmissionDate")) = admissionDate Then
            If source Is Nothing Then Set source = r
        End If
    Next r
    If source Is Nothing Then Err.Raise vbObjectError + 212, , "Admission not found"

    Dim added As ListRow
    Set added = t.ListRows.Add
    SetField t, added, "PatientID", patientId
    SetField t, added, "AdmissionDate", admissionDate
    SetField t, added, "PatientName", GetField(t, source, "PatientName")
    SetField t, added, "Room", GetField(t, source, "Room")
    SetField t, added, "AdmissionStatus", GetField(t, source, "AdmissionStatus")
    SetField t, added, "RecordID", recordId
    SetField t, added, "DischargeDate", GetField(q, job, "DischargeDate")
    SetField t, added, "Cancelled", GetField(q, job, "Cancelled")
    SetField t, added, "NoticeAt", GetField(q, job, "NoticeAt")
    SetField t, added, "Recipient", GetField(q, job, "Recipient")
    SetField t, added, "Messenger", GetField(q, job, "Messenger")
    SetField t, added, "Remarks", GetField(q, job, "Remarks")
    SetField t, added, "Signer", GetField(q, job, "Signer")
    SetField t, added, "SignedAt", GetField(q, job, "SignedAt")
End Sub

Private Sub SetField(ByVal t As ListObject, ByVal r As ListRow, ByVal columnName As String, ByVal value As Variant)
    r.Range.Cells(1, t.ListColumns(columnName).Index).Value2 = value
End Sub

Private Function GetField(ByVal t As ListObject, ByVal r As ListRow, ByVal columnName As String) As Variant
    GetField = r.Range.Cells(1, t.ListColumns(columnName).Index).Value2
End Function
